Act as: Senior IoT, ESP32, dan C++ Firmware Engineer,

Konteks Proyek:
Saya sedang membuat "Smart Machine Monitoring System (SMMS)" untuk PT. CNC menggunakan ESP32, LCD I2C 20x4, Keypad 4x4, dan MQTT. Sistem backend (Node-RED + MySQL) sudah siap.
Saya membutuhkan kode ESP32 dengan arsitektur State Machine (Finite State Machine) tanpa delay() blocking, menggunakan Preferences.h untuk menyimpan memori permanen, serta implementasi MQTT Callback untuk komunikasi Request-Response dua arah dengan Node-RED.
/iot/irfan/top8-FIX.ino  adalah kode arduino mega yang sudah jalan saat init
/iot/irfan/nanonano-FIX.ino adalah kode nano yang fix saat ini

Berikut adalah Alur UI/UX & Logika Sistem yang diimplementasikan oleh iot:

1. Boot Sequence (State: BOOT)

Saat dinyalakan, LCD 20x4 tampil: Baris 1: "    SMART MACHINE   ", Baris 2: "  MONITORING SYSTEM ", Baris 4: "       PT. CNC      ".

Tahan 3 detik (gunakan millis()), lalu tampilkan: "    Please wait.    "

2. Hidden Setup Mode (State: SETUP_MODE)

Jika saat boot tombol btnUp dan btnDown ditekan BERSAMAAN, masuk mode setup.

Layar meminta input: "Masukkan ID Mesin:". Input via keypad, tekan 'D'.

Kirim ke MQTT dan tunggu konfirmasi. Jika "Terdaftar", simpan permanen dengan <Preferences.h>.

3. Operator Login (State: LOGIN_NIK & SKILL_CHECK)

Layar utama awal meminta: "Masukkan NIK:". Input NIK, tekan 'D'. Publish MQTT, tunggu balasan (Callback).

Jika salah: "Coba Lagi".

Jika benar: Cek Skill Matrix. Jika level < 3: "Skill Matrix tidak terpenuhi". Jika >= 3: "Berhasil, Level [X]/4", lanjut input proses.

4. Input Kode Proses (State: INPUT_PROCESS & CONFIRM_PROCESS)

Layar tampil: "Masukkan Kode Proses". Input via keypad, tekan 'D'. Publish MQTT.

Server membalas dengan JSON (Nama Part, Part Number, Proses). Tampilkan di LCD (Baris 1-3).

Tekan 'D' untuk Lanjut, atau 'C' untuk Batal/Input ulang.

5. Main Monitoring Screen (State: MAIN_SCREEN & INFO_SCREEN)

Layar utama saat produksi berjalan (Gunakan blok fungsi updatePartID, updateProsesDesc, updatemcInfo, updateBarisBawah yang berbasis millis()).

Keypad: A = Ganti NIK, B = Ganti Proses, * = Info Screen (Tampil Part, No, Proses, ID Mesin).

6. Logika NG, Repair, dan Reset (State: INPUT_NG_QTY & INPUT_NG_CODE) - FITUR BARU
Berada di Main Screen, jika tombol fisik ditekan:

Jika NG - ditahan selama 5 detik: Reset seluruh counter (prodCount, NGCount, OKCount) menjadi 0.

Jika NG + ditekan (Short Press - Tambah NG):

Pindah State, Layar berubah: "Masukkan Qty NG:". Operator input angka, tekan 'D'.

Pindah State, Layar berubah: "Masukkan Kode NG:". Operator input angka/huruf, tekan 'D'.

Setelah D ditekan, kembalikan layar ke MAIN_SCREEN, tambah nilai NGCount sesuai Qty, dan kirim payload JSON via MQTT (Kategori: Tambah NG, Qty, Kode NG) ke server.

Jika NG - ditekan (Short Press - Proses Repair):

Pindah State, Layar berubah: "Masukkan Qty Repair:". Operator input angka, tekan 'D'.

Pindah State, Layar berubah: "Masukkan Kode Repair:". Operator input angka/huruf, tekan 'D'.

Setelah D ditekan, kembalikan layar ke MAIN_SCREEN, kurangi nilai NGCount sesuai Qty, dan kirim payload JSON via MQTT (Kategori: Repair NG, Qty, Kode) ke server.

7. Review Data Part per Shift (Multi-Model) - FITUR BARU
Dalam 1 shift, operator mungkin mengganti model part 3-6 kali. Sistem harus bisa menyimpan dan menampilkan history produksi setiap part dalam shift tersebut.
- Tekan 'CTRL' dari MAIN_SCREEN untuk masuk ke state REVIEW_PART_SCREEN.
- Layar menampilkan: Nama Part, No Part, Total OK, dan Total NG untuk model tersebut.
- Tekan 'PROG +' (Next) untuk melihat data model part selanjutnya yang dikerjakan di shift ini.
- Tekan 'PROG -' (Prev) untuk melihat data model part sebelumnya.
- Tekan 'CTRL' lagi (atau tombol C) untuk keluar dan kembali ke MAIN_SCREEN.
- Catatan: Data history setiap part (OK, NG, Part ID) disimpan secara non-volatile (EEPROM) agar bisa direview kapan saja selama shift belum direset.

Instruksi Tambahan untuk Struktur Kode C++:

Buat enum SystemState { BOOT, SETUP_MODE, WAIT_SETUP_REPLY, LOGIN_NIK, WAIT_NIK_REPLY, SKILL_CHECK, INPUT_PROCESS, WAIT_PROCESS_REPLY, CONFIRM_PROCESS, MAIN_SCREEN, INFO_SCREEN, INPUT_NG_QTY, INPUT_NG_CODE, REVIEW_PART_SCREEN };

Buat flag variabel seperti bool isRepairMode = false; untuk membedakan apakah State INPUT_NG_QTY dipicu oleh tombol NG+ atau NG-.

Buat fungsi void callback(char* topic, byte* payload, unsigned int length) untuk menangkap JSON MQTT.

Pastikan SEMUA pembacaan keypad dan tombol tidak menggunakan delay() blocking. Semuanya wajib pakai millis().

Cuplikan Fungsi UI Layar Utama Saya:
[PASTE-KAN KODE void updatePartID() SAMPAI void updateMcStatus() MILIK ANDA DI SINI]
Tolong berikan Full Code ESP32 (jangan sampai terpotong) yang diarsiteki dengan kokoh menggunakan FSM, non-blocking, dan siap deploy.

// ================= LOOP UTAMA =================
void loop() {
  if (!client.connected()) reconnect();
  client.loop();
  
  unsigned long currentMillis = millis();
  char key = keypad.getKey();

  switch (currentState) {
    case BOOT:
      if (currentMillis - stateStartTime < 3000) {
        // Cek hidden setup
        if (digitalRead(btnUp) == LOW && digitalRead(btnDown) == LOW) {
          changeState(SETUP_MODE);
        }
      } else {
        if (mcID == "") changeState(SETUP_MODE);
        else changeState(LOGIN_NIK);
      }
      break;

    case SETUP_MODE:
      if (key >= '0' && key <= '9' || key >= 'A' && key <= 'Z') { inputBuffer += key; forceUpdate = true; }
      else if (key == 'C' && inputBuffer.length() > 0) { inputBuffer.remove(inputBuffer.length() - 1); forceUpdate = true; }
      else if (key == 'D' && inputBuffer.length() > 0) {
        // Kirim request setup
        StaticJsonDocument<200> doc;
        doc["mcID"] = inputBuffer;
        char jsonBuffer[200]; serializeJson(doc, jsonBuffer);
        client.publish("SMMS/Request/Setup", jsonBuffer);
        changeState(WAIT_SETUP_REPLY);
      }
      if (forceUpdate) {
        lcd.clear(); lcd.setCursor(0, 0); lcd.print("Mode Setup Teknisi");
        lcd.setCursor(0, 1); lcd.print("Masukkan ID Mesin:");
        lcd.setCursor(0, 2); lcd.print(inputBuffer); lcd.blink();
        forceUpdate = false;
      }
      break;

    case WAIT_SETUP_REPLY:
    case WAIT_NIK_REPLY:
    case WAIT_PROCESS_REPLY:
      if (forceUpdate) {
        lcd.clear(); lcd.noBlink(); lcd.setCursor(4, 1); lcd.print("Memproses...");
        forceUpdate = false;
      }
      break;

    case LOGIN_NIK:
      if (key >= '0' && key <= '9' && inputBuffer.length() < 10) { inputBuffer += key; forceUpdate = true; }
      else if (key == 'C' && inputBuffer.length() > 0) { inputBuffer.remove(inputBuffer.length() - 1); forceUpdate = true; }
      else if (key == 'D' && inputBuffer.length() > 0) {
        StaticJsonDocument<200> doc;
        doc["op_NIK"] = inputBuffer;
        doc["mcID"] = mcID;
        char jsonBuffer[200]; serializeJson(doc, jsonBuffer);
        client.publish("SMMS/Request/Login", jsonBuffer);
        changeState(WAIT_NIK_REPLY);
      }
      if (forceUpdate) {
        lcd.clear(); lcd.setCursor(0, 0); lcd.print("PT. CNC - " + mcID);
        lcd.setCursor(0, 1); lcd.print("Masukkan NIK:");
        lcd.setCursor(0, 2); lcd.print(inputBuffer); lcd.blink();
        forceUpdate = false;
      }
      break;

    case SKILL_CHECK:
      // Tampilan transisi sukses/gagal dari callback
      if (currentMillis - stateStartTime > 3000) {
        if (op_Level >= 3) changeState(INPUT_PROCESS);
        else changeState(LOGIN_NIK);
      } else if (forceUpdate) {
        lcd.clear(); lcd.noBlink();
        if (op_Level >= 3) {
          lcd.setCursor(0, 0); lcd.print("Berhasil!");
          lcd.setCursor(0, 1); lcd.print(op_Name.substring(0, 20));
          lcd.setCursor(0, 2); lcd.print("Level: " + String(op_Level) + "/4");
        } else {
          lcd.setCursor(0, 0); lcd.print(tempMessage); // Pesan Error
          lcd.setCursor(0, 2); lcd.print("Tekan D coba lagi");
        }
        forceUpdate = false;
      }
      if (op_Level < 3 && key == 'D') changeState(LOGIN_NIK);
      break;

    case INPUT_PROCESS:
      if (key >= '0' && key <= '9' || key >= 'A' && key <= 'Z') { inputBuffer += key; forceUpdate = true; }
      else if (key == 'C' && inputBuffer.length() > 0) { inputBuffer.remove(inputBuffer.length() - 1); forceUpdate = true; }
      else if (key == 'D' && inputBuffer.length() > 0) {
        StaticJsonDocument<200> doc;
        doc["kode_proses"] = inputBuffer;
        char jsonBuffer[200]; serializeJson(doc, jsonBuffer);
        client.publish("SMMS/Request/Proses", jsonBuffer);
        changeState(WAIT_PROCESS_REPLY);
      }
      if (forceUpdate) {
        lcd.clear(); lcd.setCursor(0, 0); lcd.print("Operator: " + op_NIK);
        lcd.setCursor(0, 1); lcd.print("Masukkan Kd Proses:");
        lcd.setCursor(0, 2); lcd.print(inputBuffer); lcd.blink();
        forceUpdate = false;
      }
      break;

    case CONFIRM_PROCESS:
      if (forceUpdate) {
        lcd.clear(); lcd.noBlink();
        lcd.setCursor(0, 0); lcd.print(partID.substring(0, 20));
        lcd.setCursor(0, 1); lcd.print(partNo.substring(0, 20));
        lcd.setCursor(0, 2); lcd.print(prosesDesc.substring(0, 20));
        lcd.setCursor(0, 3); lcd.print("D:Lanjut  C:Batal");
        forceUpdate = false;
      }
      if (key == 'D') changeState(MAIN_SCREEN);
      else if (key == 'C') changeState(INPUT_PROCESS);
      break;

    case MAIN_SCREEN:
      if (forceUpdate) { lcd.clear(); lcd.noBlink(); lastPartID = ""; lastProsesDesc = ""; forceUpdate = false; }
      
      // Navigasi Layar Utama
      if (key == 'A') changeState(LOGIN_NIK);
      else if (key == 'B') changeState(INPUT_PROCESS);
      else if (key == '*') changeState(INFO_SCREEN);

      readMachineStatus();
      readProductionButtons();
      processMainScreen(); // Update UI LCD (Teks Berjalan dll)

      // Publish Realtime Data tiap 2 Detik
      if (currentMillis - lastMsgTime > 2000) {
        lastMsgTime = currentMillis;
        StaticJsonDocument<512> doc;
        doc["mcID"] = mcID;
        doc["op_NIK"] = op_NIK;
        doc["partID"] = partID;
        doc["kode_proses"] = kode_proses;
        doc["mcStatus"] = mcStatus;
        doc["mcInfo"] = mcInfo;
        doc["prodCount"] = prodCount;
        doc["OKCount"] = OKCount;
        doc["NGCount"] = NGCount;
        
        char jsonStr[512]; serializeJson(doc, jsonStr);
        client.publish("SMMS/Data/Produksi", jsonStr);
      }
      break;

    case INFO_SCREEN:
      if (forceUpdate) {
        lcd.clear(); lcd.noBlink();
        lcd.setCursor(0, 0); lcd.print(partID.substring(0, 20));
        lcd.setCursor(0, 1); lcd.print(partNo.substring(0, 20));
        lcd.setCursor(0, 2); lcd.print(prosesDesc.substring(0, 20));
        lcd.setCursor(0, 3); lcd.print("ID Mesin: " + mcID);
        forceUpdate = false;
      }
      if (key == '*') changeState(MAIN_SCREEN); // Kembali ke utama
      break;
  }
}

// ================= CALLBACK MQTT (Terima Data Node-RED) =================
void callback(char* topic, byte* payload, unsigned int length) {
  String topicStr = String(topic);
  StaticJsonDocument<512> doc;
  deserializeJson(doc, payload, length);
  String status = doc["status"].as<String>();

  if (topicStr == "SMMS/Reply/Setup" && currentState == WAIT_SETUP_REPLY) {
    if (status == "registered") {
      mcID = inputBuffer;
      preferences.putString("mcID", mcID);
      lcd.clear(); lcd.print("Berhasil Disimpan!");
      delay(1500); // Exception delay untuk pesan fix memory
      changeState(LOGIN_NIK);
    } else {
      lcd.clear(); lcd.print("ID Gagal Terdaftar!");
      delay(1500); changeState(SETUP_MODE);
    }
  }
  
  else if (topicStr == "SMMS/Reply/Login" && currentState == WAIT_NIK_REPLY) {
    if (status == "success") {
      op_NIK = inputBuffer;
      op_Name = doc["nama"].as<String>();
      op_Level = doc["level"].as<int>();
      if (op_Level < 3) tempMessage = "Skill tdk terpenuhi";
    } else {
      op_Level = 0; tempMessage = "NIK Tidak Terdaftar";
    }
    changeState(SKILL_CHECK);
  }
  
  else if (topicStr == "SMMS/Reply/Proses" && currentState == WAIT_PROCESS_REPLY) {
    if (status == "found") {
      kode_proses = inputBuffer;
      partID = doc["nama_part"].as<String>();
      partNo = doc["part_no"].as<String>();
      prosesDesc = doc["nama_proses"].as<String>();
      
      // Tambah spasi padding untuk teks berjalan
      while(partID.length() < 20) partID += " ";
      while(prosesDesc.length() < 20) prosesDesc += " ";
      changeState(CONFIRM_PROCESS);
    } else {
      lcd.clear(); lcd.print("Kode Tidak Valid!");
      delay(1500); changeState(INPUT_PROCESS);
    }
  }
}

// ================= MANAJEMEN STATE =================
void changeState(SystemState newState) {
  currentState = newState;
  stateStartTime = millis();
  inputBuffer = "";
  forceUpdate = true;
  lcd.noBlink();

  if (newState == BOOT) {
    lcd.clear();
    lcd.setCursor(4, 0); lcd.print("SMART MACHINE");
    lcd.setCursor(2, 1); lcd.print("MONITORING SYSTEM");
    lcd.setCursor(7, 3); lcd.print("PT. CNC");
  } 
  else if (newState == MAIN_SCREEN) {
    posisi1 = 0; posisi2 = 0; posisi3 = 0;
  }
}

// ================= LOGIKA TOMBOL & COUNTER (TERMASUK RESET 5 DETIK) =================
void readProductionButtons() {
  unsigned long currentMillis = millis();

  // Tombol Count
  bool currentCountState = digitalRead(btnCount);
  if (lastCountState == HIGH && currentCountState == LOW) { 
    prodCount++; forceUpdate = true; delay(30); // mini debounce
  }
  lastCountState = currentCountState;

  // Tombol NG+
  bool currentNgAddState = digitalRead(ngAdd);
  if (lastNgAddState == HIGH && currentNgAddState == LOW && NGCount < prodCount) { 
    NGCount++; forceUpdate = true; delay(30); 
  }
  lastNgAddState = currentNgAddState;

  // Tombol NG- (Pendek = Kurang, Tahan 5D = Reset)
  int ngSubState = digitalRead(ngSub);
  if (ngSubState == LOW) { // Tombol sedang ditekan
    if (!ngSubPressed) {
      ngSubPressed = true;
      ngSubPressTime = currentMillis;
    } else if (currentMillis - ngSubPressTime >= 5000) {
      // RESET ALL COUNTER JIKA DITAHAN 5 DETIK
      prodCount = 0; NGCount = 0; OKCount = 0;
      lcd.clear(); lcd.setCursor(2,1); lcd.print("COUNTER DIRESET!");
      delay(1500); // Feedback UI
      forceUpdate = true;
      ngSubPressTime = currentMillis + 99999; // cegah trigger berulang
    }
  } else { // Tombol dilepas
    if (ngSubPressed) {
      // Jika dilepas sebelum 5 detik (dan bukan bounce)
      if (currentMillis - ngSubPressTime < 5000 && currentMillis - ngSubPressTime > 50) {
        if (NGCount > 0) NGCount--;
        forceUpdate = true;
      }
      ngSubPressed = false;
    }
  }
}

// ================= LOGIKA STATUS MESIN =================
void readMachineStatus() {
  if (digitalRead(mcON) == LOW) { 
    mcStatus = "off"; mcInfo = "Mesin Off"; 
  } 
  else if (digitalRead(alarmPin) == HIGH) { 
    mcStatus = "alarm"; mcInfo = "Mesin Alarm"; 
  } 
  else if (digitalRead(mcRun) == HIGH) { 
    mcStatus = "run"; mcInfo = "Mesin Running"; 
  } 
  else {
    mcStatus = "standBy";
    if (previousMcStatus != "standBy") previousMcInfo = "Stand By";
    
    String currentMcInfo = previousMcInfo;
    if (digitalRead(btnDandory) == LOW) currentMcInfo = "Dandory";
    else if (digitalRead(btnMinum) == LOW) currentMcInfo = "Minum";
    
    mcInfo = currentMcInfo;
    previousMcInfo = currentMcInfo;
  }
  previousMcStatus = mcStatus;
}

// ================= FUNGSI LAYAR UTAMA =================
void processMainScreen() {
  unsigned long currentMillis = millis();
  
  // Update Part ID (Baris 0)
  if (partID.length() > 20) {
    if (currentMillis - previousMillis1 >= intervalScroll) {
      previousMillis1 = currentMillis;
      lcd.setCursor(0, 0); lcd.print("                    ");
      lcd.setCursor(0, 0); lcd.print(partID.substring(posisi1, posisi1 + 20));
      posisi1 += 5;
      if (posisi1 >= partID.length() - 15) posisi1 = 0;
    }
  } else if (partID != lastPartID || forceUpdate) {
    lcd.setCursor(0, 0); lcd.print("                    ");
    lcd.setCursor(0, 0); lcd.print(partID);
    lastPartID = partID;
  }

  // Update Proses Desc (Baris 1)
  if (prosesDesc.length() > 20) {
    if (currentMillis - previousMillis2 >= intervalScroll) {
      previousMillis2 = currentMillis;
      lcd.setCursor(0, 1); lcd.print("                    ");
      lcd.setCursor(0, 1); lcd.print(prosesDesc.substring(posisi2, posisi2 + 20));
      posisi2 += 5;
      if (posisi2 >= prosesDesc.length() - 15) posisi2 = 0;
    }
  } else if (prosesDesc != lastProsesDesc || forceUpdate) {
    lcd.setCursor(0, 1); lcd.print("                    ");
    lcd.setCursor(0, 1); lcd.print(prosesDesc);
    lastProsesDesc = prosesDesc;
  }

  // Update Mc Info (Baris 2)
  String displayInfo = mcInfo;
  while(displayInfo.length() < 20) displayInfo += " "; // padding
  if (displayInfo.length() > 20) {
    if (currentMillis - previousMillis3 >= intervalScroll) {
      previousMillis3 = currentMillis;
      lcd.setCursor(0, 2); lcd.print("                    ");
      lcd.setCursor(0, 2); lcd.print(displayInfo.substring(posisi3, posisi3 + 20));
      posisi3 += 5;
      if (posisi3 >= displayInfo.length() - 15) posisi3 = 0;
    }
  } else if (displayInfo != lastmcInfo || forceUpdate) {
    lcd.setCursor(0, 2); lcd.print("                    ");
    lcd.setCursor(0, 2); lcd.print(displayInfo);
    lastmcInfo = displayInfo;
  }

  // Update Baris Bawah (Baris 3) - NIK & Counter
  if (op_NIK != lastop_NIK || forceUpdate) {
    lcd.setCursor(0, 3); lcd.print("         ");
    lcd.setCursor(0, 3); lcd.print(op_NIK);
    lastop_NIK = op_NIK;
  }

  OKCount = prodCount - NGCount;
  if (OKCount != lastOKCount || forceUpdate) {
    lcd.setCursor(9, 3); lcd.print("|      ");
    lcd.setCursor(11, 3); lcd.print(OKCount);
    lastOKCount = OKCount;
  }

  if (NGCount != lastNGCount || forceUpdate) {
    lcd.setCursor(16, 3); lcd.print("/   ");
    lcd.setCursor(17, 3); lcd.print(NGCount);
    lastNGCount = NGCount;
  }
}

// ================= SETUP WIFI & MQTT =================
void setup_wifi() {
  lcd.setCursor(0, 0); lcd.print("Koneksi WiFi...     ");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
  lcd.setCursor(0, 0); lcd.print("WiFi Terhubung!     ");
  delay(1000);
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32_Mesin_" + String(random(0xffff), HEX);
    client.setBufferSize(1024);
    
    // Subscribe ke semua topik Reply Node-RED saat berhasil konek
    if (client.connect(clientId.c_str())) {
      client.subscribe("SMMS/Reply/Setup");
      client.subscribe("SMMS/Reply/Login");
      client.subscribe("SMMS/Reply/Proses");
    } else { 
      delay(5000); 
    }
  }
}
