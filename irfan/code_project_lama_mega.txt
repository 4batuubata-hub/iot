void loop() {
/*char key = keypad.getKey();
  if (key) {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("Key Pressed:");
    lcd.setCursor(0,1);
    lcd.print(key);
  }
*/
  if (!client.connected()) {
    digitalWrite(brokerLed, LOW);
    unsigned long now = millis();
    if (now - lastReconnectAttempt > 5000) {
      lastReconnectAttempt = now;
      reconnect();
    }
  } else {
    digitalWrite(brokerLed, HIGH);
    client.loop();
  }

  if (millis() - lastPublish > publishInterval) {
    lastPublish = millis();
    publishData();
    printData();
  }

  handleHTTP(); // Cek dan respon jika ada HTTP request
  
  //HANDLING COUNTER DARI NANO
  cycleCountRecieved = readCounterFromNano();

  //HANDLING PERHITUNGAN HASIL PRODUKSI
  cycleCount = cycleCountSaved + cycleCountRecieved;
  data.cycleCount = cycleCount;
  prodCount = cycleCount * cavity;

  //HANDLING mcStatus & mcInfo
  updateMcStatus();

  //HANDLING TIMESTAMP
  DateTime now = rtc.now();
  snprintf(timestamp, sizeof(timestamp), "%04d-%02d-%02dT%02d:%02d:%02d",
             now.year(), now.month(), now.day(), now.hour(), now.minute(), now.second());

  //HANDLING IDENTITAS PART
  unsigned long currentMillis = millis();
  bool currentStateUp = digitalRead(PB_UP);
  bool currentStateDown = digitalRead(PB_DOWN);
  bool currentStateOk = digitalRead(PB_OK);

  if (currentMillis - lastButtonTime > buttonDebounceDelay) {
      if (lastStateUp == HIGH && currentStateUp == LOW && currentStateOk == LOW) {
          data.selectCode++;
          if (data.selectCode > 9) data.selectCode = 9;
          lastButtonTime = currentMillis;
          dataChanged = true;
      }
      if (lastStateDown == HIGH && currentStateDown == LOW && currentStateOk == LOW) {
          data.selectCode--;
          if (data.selectCode < 0) data.selectCode = 0;
          lastButtonTime = currentMillis;
          dataChanged = true;
      }
  }

  lastStateUp = currentStateUp;
  lastStateDown = currentStateDown;
  lastStateOk = currentStateOk;

  if (dataChanged) {
      updateData(data.selectCode);
      dataChanged = false;
  }

/*
  //HANDLING LAYAR BERKEDIP
  if (mcStatus != "run" && !inputEnabled && !showSavedMessage) {
      if (currentMillis - previousMillis >= 500) {
          previousMillis = currentMillis;
          backlightState = !backlightState;
          if (backlightState) {
              lcd.backlight();
          } else {
              lcd.noBacklight();
          }
      }
  } else {
      lcd.backlight();
  }
*/

  //HANDLING LED BERKEDIP
  if (mcStatus != "run") {
      if (currentMillis - previousMillis >= 250) {
          previousMillis = currentMillis;
          ledState = !ledState;
          if (ledState) {
              digitalWrite(stopLed, HIGH);
          } else {
              digitalWrite(stopLed, LOW);
          }
      }
  } else {
      digitalWrite(stopLed, LOW);
  }

  //HANDLING RESET VARIABLE
  if (digitalRead(ngSub) == LOW) {  // Tombol ditekan (aktif LOW)
      if (!isPressed) {  // Jika tombol baru saja ditekan
          pressStartTime = millis();  // Catat waktu awal tekan
          isPressed = true;
      }
      else if (millis() - pressStartTime >= 5000) {  // Jika tombol ditekan selama 5 detik
          resetData();  // Panggil fungsi reset
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("Data berhasil");
          lcd.setCursor(0, 1);
          lcd.print("di reset.");
          showSavedMessage = true;  // Aktifkan flag untuk menampilkan pesan
          nikSavedTime = millis();  // Catat waktu saat pesan ditampilkan
      }
  } else {  // Tombol dilepas
      isPressed = false;
  }

  //HANDLING KEYPAD NIK
  key = keypad.getKey();
  if (key) {
      // Mode Input NIK (Tombol "A" untuk masuk ke mode input)
      if (key == 'A') {
          inputEnabled = true;
          data.op_NIK = "";
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("Silahkan masukan");
          lcd.setCursor(0, 1);
          lcd.print("NIK anda:");
          key = 0;
      }
  }

    // Jalankan fungsi input NIK jika mode aktif
  if (inputEnabled) {
    handleNIKInput();
  }

 if (showSavedMessage && millis() - nikSavedTime >= 1500) {
    lcd.clear();
    showSavedMessage = false;
    forceUpdate = true;
    isPressed = false;
    }

  //HANDLING MAIN SCREEN
  if (!inputEnabled && !showSavedMessage) {
    updatePartID();
    updateProsesDesc();
    updatemcInfo();
    updateOperatorNIK();
    updateOKCount();
    updateNGCount();
    forceUpdate = false;
  }

  //HANDLING COUNTER NG
  updateNg();

  //HANDLING PRINT DATA KE SERIAL & SAVE EEPROM
  if (currentMillis - lastSerialTime > 2000) {
      saveDataToEEPROM(data);
      lastSerialTime = currentMillis;
  }

  //HANDLING HITUNG LOSS TIME
  hitungLoss();

}

void handleNIKInput() {
  if (key) {
    if (key >= '0' && key <= '9') {
      if (data.op_NIK.length() < 8) {
        data.op_NIK += key;
        lcd.setCursor(0, 2);
        lcd.print(data.op_NIK);
      } else {
        lcd.setCursor(0, 3);
        lcd.print("                    ");
        lcd.setCursor(0, 3);
        lcd.print("Maks 8 digit!");
      }
    } else if (key == 'B') { // Tombol backspace
      if (data.op_NIK.length() > 0) {
        data.op_NIK.remove(data.op_NIK.length() - 1);
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Silahkan masukan");
        lcd.setCursor(0, 1);
        lcd.print("NIK anda:");
        lcd.setCursor(0, 2);
        lcd.print(data.op_NIK);
      }
    } else if (key == 'D') {
      if (data.op_NIK.length() == 8) {
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("NIK tersimpan");
        lcd.setCursor(0, 1);
        lcd.print(data.op_NIK);
        showSavedMessage = true;
		    inputEnabled = false;
        nikSavedTime = millis();
      } else {
        lcd.setCursor(0, 3);
        lcd.print("                    ");
        lcd.setCursor(0, 3);
        lcd.print("NIK harus 8 digit!");
      }
    } else {
      lcd.setCursor(0, 3);
      lcd.print("                    ");
      lcd.setCursor(0, 3);
      lcd.print("Input tidak valid!");
    }
  }
}

void updatePartID() {
    static String lastPartID = "";
    if (partID.length() > 20) {
        unsigned long currentMillis = millis();
        if (currentMillis - previousMillis1 >= intervalScroll) {
            previousMillis1 = currentMillis;
            lcd.setCursor(0, 0);
            lcd.print("                    ");
            lcd.setCursor(0, 0);
            lcd.print(partID.substring(posisi1, posisi1 + 20));
            posisi1 += 5;
            if (posisi1 >= partID.length() - 15) posisi1 = 0;
        }
    } else if (partID != lastPartID || forceUpdate) {
        lcd.setCursor(0, 0);
        lcd.print("                    ");
        lcd.setCursor(0, 0);
        lcd.print(partID);
        lastPartID = partID;
    }
}

void updateMcStatus() {
    bool mcOnState = digitalRead(mcON) == LOW;
    bool alarmState = digitalRead(alarmPin) == HIGH;
    bool mcRunState = digitalRead(mcRun) == HIGH;

    if (!mcOnState) {
        mcStatus = "off";
        mcInfo = "Mesin Off";
    } else if (!alarmState) {
        mcStatus = "alarm";
        mcInfo = "Mesin Alarm";
    } else if (!mcRunState) {
        mcStatus = "run";
        mcInfo = "Mesin Running";
    } else {
        mcStatus = "standBy";
        if (previousMcStatus != "standBy") {
            previousMcInfo = "Stand By";
        }
        checkAndUpdateMcInfo();
    }
    previousMcStatus = mcStatus;
}

void updateNg() {
  unsigned long currentTime = millis();
  bool currentStateAdd = digitalRead(ngAdd);
  bool currentStateSub = digitalRead(ngSub);

  if (currentTime - lastDebounceTime > debounceDelay) {
      if (lastStateNgAdd == HIGH && currentStateAdd == LOW) {
          if (data.NGCount < prodCount) {
                data.NGCount++;
                lastDebounceTime = currentTime;
          }
      }

      if (lastStateNgSub == HIGH && currentStateSub == LOW) {
          if (data.NGCount > 0) {
              data.NGCount--;
              lastDebounceTime = currentTime;
          }
      }
    }

  lastStateNgAdd = currentStateAdd;
  lastStateNgSub = currentStateSub;
}

// ========== EEPROM HELPER FUNCTIONS ==========
void writeStringToEEPROM(int addr, String data) {
    int len = data.length();
    for (int i = 0; i < len; i++) {
        Wire.beginTransmission(EEPROM_ADDRESS);
        Wire.write((int)(addr >> 8));
        Wire.write((int)(addr & 0xFF));
        Wire.write(data[i]);
        Wire.endTransmission();
        delay(5);
        addr++;
    }
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((int)(addr >> 8));
    Wire.write((int)(addr & 0xFF));
    Wire.write('\0');  // Null terminator
    Wire.endTransmission();
    delay(5);
}

String readStringFromEEPROM(int addr) {
    char data[32];
    int len = 0;
    while (len < 31) {
        Wire.beginTransmission(EEPROM_ADDRESS);
        Wire.write((int)(addr >> 8));
        Wire.write((int)(addr & 0xFF));
        Wire.endTransmission();
        Wire.requestFrom(EEPROM_ADDRESS, 1);
        if (Wire.available()) {
            char c = Wire.read();
            if (c == '\0') break;
            data[len++] = c;
        }
        addr++;
    }
    data[len] = '\0';
    return String(data);
}


void writeByteToEEPROM(int addr, byte data) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.write(data);
    Wire.endTransmission();
    delay(5);
}

byte readByteFromEEPROM(int addr) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.endTransmission();
    Wire.requestFrom(EEPROM_ADDRESS, 1);
    return Wire.available() ? Wire.read() : 0;
}

void writeIntToEEPROM(int addr, int data) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.write((data >> 8) & 0xFF);
    Wire.write(data & 0xFF);
    Wire.endTransmission();
    delay(5);
}

int readIntFromEEPROM(int addr) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.endTransmission();
    Wire.requestFrom(EEPROM_ADDRESS, 2);
    int data = (Wire.read() << 8) | Wire.read();
    return data;
}

void writeLongToEEPROM(int addr, long data) {
    // Hapus data dulu sebelum menulis (overwrite dengan 0)
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.write(0x00);
    Wire.write(0x00);
    Wire.write(0x00);
    Wire.write(0x00);
    Wire.endTransmission();
    delay(10); // Tambah delay

    // Tulis data baru
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.write((data >> 24) & 0xFF);
    Wire.write((data >> 16) & 0xFF);
    Wire.write((data >> 8) & 0xFF);
    Wire.write(data & 0xFF);
    Wire.endTransmission();
    delay(10); // Tambah delay
}


long readLongFromEEPROM(int addr) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.endTransmission();

    Wire.requestFrom(EEPROM_ADDRESS, 4);
    if (Wire.available() < 4) {
//        Serial.println("Error: EEPROM Read failed!");
        return 0;
    }

    byte b1 = Wire.read();
    byte b2 = Wire.read();
    byte b3 = Wire.read();
    byte b4 = Wire.read();

    long data = ((long)b1 << 24) | ((long)b2 << 16) | ((long)b3 << 8) | b4;
    return data;
}

// Fungsi membaca inital flag
byte readEEPROM(uint16_t addr) {
    Wire.beginTransmission(EEPROM_ADDRESS);
    Wire.write((addr >> 8) & 0xFF);
    Wire.write(addr & 0xFF);
    Wire.endTransmission();

    Wire.requestFrom(EEPROM_ADDRESS, 1);
    if (Wire.available()) {
        return Wire.read();
    }
    return 0xFF; // Default jika tidak ada data
}

// Fungsi untuk membaca counter dari Arduino Nano
unsigned int readCounterFromNano() {
  Wire.requestFrom(8, 2); // Minta 2 byte dari Arduino Nano dengan alamat 8
  if (Wire.available() == 2) {
    return (Wire.read() << 8) | Wire.read(); // Gabungkan byte tinggi dan rendah
  }
}

void handleHTTP() {
  EthernetClient clientHTTP = serverHTTP.available();
  if (clientHTTP) {
    Serial.println("?? HTTP client terhubung");

    while (clientHTTP.connected()) {
      if (clientHTTP.available()) {
        clientHTTP.readStringUntil('\r'); // Request line
        clientHTTP.read(); // newline

        StaticJsonDocument<896> doc;
        fillJson(doc);

        // Header HTTP
        clientHTTP.println("HTTP/1.1 200 OK");
        clientHTTP.println("Content-Type: text/html");
        clientHTTP.println("Connection: close");
        clientHTTP.println();

        // Kirim HTML stream per bagian
        clientHTTP.println("<!DOCTYPE html><html><head><meta charset='UTF-8'>");
        clientHTTP.println("<title>Monitoring Mesin</title>");
        clientHTTP.println("<style>body{font-family:Arial;}table{border-collapse:collapse;width:100%;}th,td{border:1px solid #ccc;padding:8px;text-align:left;}th{background:#eee;}</style></head><body>");
        clientHTTP.println("<h2>Status Mesin</h2><table>");

        for (JsonPair kv : doc.as<JsonObject>()) {
          clientHTTP.print("<tr><th>");
          clientHTTP.print(kv.key().c_str());
          clientHTTP.print("</th><td>");
          clientHTTP.print(kv.value().as<String>());
          clientHTTP.println("</td></tr>");
        }

        clientHTTP.println("</table></body></html>");
        break;
      }
    }

    delay(1);
    clientHTTP.stop();
    Serial.println("?? HTTP client disconnected");
  }
}
